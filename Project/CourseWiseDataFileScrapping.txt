import requests
import json
import os

# Create folder to store course files
os.makedirs("courses_data", exist_ok=True)

url = "https://api.sunbeaminfo.in/api/v1/courses?page=1&limit=100&domain=sunbeaminfo.in"

response = requests.get(url)
resp = response.json()

if resp["success"]:
    data = resp["data"]

    for course in data:
        slug = course["slug"]
        course_name = course["name"]

        print(f"Processing: {course_name}")

        course_url = f"https://api.sunbeaminfo.in/api/v1/courses/{slug}?domain=sunbeaminfo.in"

        course_response = requests.get(course_url)
        course_json = course_response.json()

        # Create safe filename
        filename = "".join(c if c.isalnum() else "_" for c in course_name)

        # Save JSON into txt file
        with open(f"courses_data/{filename}.txt", "w", encoding="utf-8") as f:
            json.dump(course_json, f, indent=4, ensure_ascii=False)

        print(f"Saved: {filename}.txt")
        print("-" * 50)

else:
    print("Failed to Request")
